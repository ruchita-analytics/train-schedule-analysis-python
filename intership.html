Task 2.2 – Compute Total Journey Duration¶

Purpose: To calculate the total journey duration for each train using its departure and arrival times.¶
In [39]:
import pandas as pd

# Step 1: Convert Arrival Time
df['Arrival_time'] = pd.to_datetime(
    df['Arrival_time'].astype(str),
    format='%H:%M:%S',
    errors='coerce'
)

# Step 2: Convert Departure Time
df['Departure_Time'] = pd.to_datetime(
    df['Departure_Time'].astype(str),
    format='%H:%M:%S',
    errors='coerce'
)

# Step 3: Sort data
df_sorted = df.sort_values(['Train_No', 'SN'])

# Step 4: Group by Train
train_group = df_sorted.groupby('Train_No')

# Step 5: First departure time
start_time = train_group['Departure_Time'].first()

# Step 6: Last arrival time
end_time = train_group['Arrival_time'].last()

# Step 7: Create table
train_times = pd.DataFrame({
    'Start_Time': start_time,
    'End_Time': end_time
})

# Step 8: Reset index
train_times = train_times.reset_index()

# Step 9: Calculate journey duration
train_times['Journey_Duration'] = (
    train_times['End_Time'] - train_times['Start_Time']
)

# Step 10: Identify overnight journeys
overnight = train_times['Journey_Duration'] < pd.Timedelta(0)

# Step 11: Correct overnight journeys
train_times.loc[
    overnight,
    'Journey_Duration'
] += pd.Timedelta(days=1)

# Step 12: Convert duration into hours
train_times['Journey_Hours'] = (
    train_times['Journey_Duration'].dt.total_seconds() / 3600
)

# Step 13: Display result
print(train_times.head(10))

import pandas as pd

# Step 1: Sort data
df_sorted = df.sort_values(['Train_No', 'SN'])

# Step 2: Find source and destination for each train
train_routes = df_sorted.groupby('Train_No').agg(
    Source_Station=('Station_Name', 'first'),
    Destination_Station=('Station_Name', 'last')
).reset_index()

# Step 3: Add journey duration
train_routes = pd.merge(
    train_routes,
    train_times[['Train_No', 'Journey_Hours']],
    on='Train_No',
    how='left'
)

# Step 4: Create enquiry function
def train_enquiry(source, destination):

    result = train_routes[
        (train_routes['Source_Station'].str.lower() == source.lower()) &
        (train_routes['Destination_Station'].str.lower() == destination.lower())
    ]

    return result

# Step 5: User input
source = input("Enter Source Station: ")
destination = input("Enter Destination Station: ")

# Step 6: Search trains
result = train_enquiry(source, destination)

# Step 7: Display result
if result.empty:

    print("\nNo direct train found.")

else:

    print("\nAvailable Direct Trains:")

    print(
        result[
            ['Train_No',
             'Source_Station',
             'Destination_Station',
             'Journey_Hours']
        ].to_string(index=False)
    )
